    public static <E> Elem<E> concatIterative(final Elem<E> first, final Elem<E> second)
    {
      if (first == null){
          return second;
      }
      Elem<E> last = first;
      for(; last.getNext() != null; last = last.getNext());
      last.setNext(second);
      return first;
    }
