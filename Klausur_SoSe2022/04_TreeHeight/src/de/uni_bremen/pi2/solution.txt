package de.uni_bremen.pi2;

/**
 * Die Klasse enthält Operationen auf Binärbäumen.
 */
public class Tree
{
    /**
     * Bestimmt die Höhe eines Binärbaums, d.h. die Anzahl der Knoten auf dem
     * längsten Weg von der Wurzel zu einem Blatt. Ein leerer Baum hat die Höhe
     * 0, ein Baum aus nur einer Wurzel die Höhe 1.
     * @param root Die Wurzel des Baums. Ist null, wenn der Baum leer ist.
     * @param <E> Der Typ der in den Knoten gespeicherten Werte.
     * @return Die Höhe des Baums.
     */
    public static <E> int treeHeight(final Node<E> root)
    {

        if(root == null) {
            return 0;
        }


        int leftSideCount = treeHeight(root.getLeft());
        int rightSideCount = treeHeight(root.getRight());


         return 1 + Math.max(leftSideCount, rightSideCount);
    }
}
